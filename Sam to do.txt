data class UserRewards(
    val userId: String,
    val totalPoints: Int,
    val redeemedPoints: Int,
    val level: String // Bronze, Silver, Gold, Platinum
)

// Add redemption logic
fun redeemPoints(userId: String, points: Int) {
    val userRewards = repository.getUserRewards(userId)
    if (userRewards.totalPoints >= points) {
        val discountCoupon = generateCouponCode(points)
        repository.updateUserRewards(userId, userRewards.copy(
            totalPoints = userRewards.totalPoints - points,
            redeemedPoints = userRewards.redeemedPoints + points
        ))
        return discountCoupon
    }
}

D. डिपेंडन्सीज इन्स्टॉल करणे (Dependencies Setup)
🔹 Android App (Kotlin) साठी:

# Android Studio टर्मिनलमध्ये रन करा
cd D:\VGAS_Project
# Android Studio स्वतःच डिपेंडन्सीज इन्स्टॉल करेल
# जर योग्य नाही तर:
# File → Settings → Build, Execution, Deployment → Gradle → "Use Gradle From"
🔹 Backend (Python) साठी:

cd D:\VGAS_Project\backend
pip install -r requirements.txt
playwright install chromium
🔹 WhatsApp Bot (Node.js) साठी:

cd D:\VGAS_Project\bot
npm install
🚀 तिसरा भाग: प्रोजेक्ट रन करणे आणि चाचणी करणे
📱 १. Android ऍप रन करणे (Emulator वर):
Android Studio मध्ये प्रोजेक्ट उघडा
Top Toolbar मध्ये "Device Manager" (📱) वर क्लिक करा
"Create Virtual Device" → Pixel 6 Pro निवडा → "Next" → "Finish"
"Run" (▶️) बटण वर क्लिक करा
Emulator उघडेल आणि VGAS Shopping Pro ऍप १-२ मिनिटांत इन्स्टॉल होईल
✅ Test करणे:

Amazon/Flipkart लिंक पेस्ट करा
"Analyze Price" वर क्लिक करा
प्राइस तुलना, फेक डिस्काउंट डिटेक्शन, चॅट असिस्टंट पाहा
🖥️ २. Backend Server चालू करणे:
cd D:\VGAS_Project\backend
uvicorn app.main:app --reload --port 8000
✅ Test करणे:

ब्राउझरमध्ये http://localhost:8000/docs उघडा
Swagger UI मध्ये API टेस्ट करा
💬 ३. WhatsApp Bot चालू करणे:
cd D:\VGAS_Project\bot
npm start
✅ Test करणे:

Terminal मध्ये QR Code दिसेल
WhatsApp → Linked Devices → Link a Device वर जा
QR Code स्कॅन करा
Amazon/Flipkart लिंक पाठवा → ऍनालिसिस रिजल्ट मिळेल
